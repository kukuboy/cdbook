# 使用方法

## 关键文件dialog.js

    /*
    * @Description:
    * @Author: 水印红枫
    * @Date: 2020-07-02 16:46:07
    * @LastEditors: 水印红枫
    */

    import Vue from "vue";

    import Toast from "./toast";

    const Dialog = {
    install(Vue) {
        Object.defineProperty(Vue.prototype, "$Dialog", {
        value: this,
        writable: false
        });
    },
    Toast(options = {}) {
        const ToastFrame = Vue.extend(Toast);
        let vm = document.getElementById("toast");
        if (vm !== null) {
        document.body.removeChild(vm);
        }
        let div = document.createElement("div");
        div.id = "Toast";
        document.body.appendChild(div);
        let $vm = new ToastFrame({
        el: "#Toast"
        });
        if (typeof options === "object") {
        Object.assign($vm, options);
        } else if (typeof options === "string" || typeof options === "number") {
        $vm.content = options;
        }
    },
    }
    export default Dialog;

## 引入方法
    __在main.js中使用vue.use即可__

    import Vue from "vue";
    import dialog from "@/util/dialog/dialog";
    Vue.use(dialog);

## 组件相关更改

    接受数据同样使用props

    props:{
        data:{
            type: Object,
            default:()=>{
                return {text : ""}
            }
        },
        id:{
            type: String,
            default:"111"
        },
        dialogAction: {
            type: Function,
            default: value => {
                return value;
            }
        }
    }

    销毁不再使用v-show或v-if,使用

    close() {
      this.dialogAction("close");
      let vm = document.getElementById("Toast");
      if (vm !== null) {
        document.body.removeChild(vm);
      }
    },

## 组件使用

    this.$Dialog.Toast({
        data:{},
        id:"222",
        dialogAction: val => {
            console.log(val)
        }
    })

    该函数可接受关闭回馈以及组件向外传递数据
